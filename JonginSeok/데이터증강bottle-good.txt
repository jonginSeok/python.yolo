# 1단계: 증강 이미지 생성 스크립트
# ===================================== #
import os
from PIL import Image
from torchvision import transforms
from tqdm import tqdm

# 증강 파이프라인 정의
bottle_good_aug = transforms.Compose([
    transforms.RandomRotation(degrees=180),
    transforms.ColorJitter(brightness=0.2, contrast=0.2, saturation=0.1),
    transforms.CenterCrop(size=(400, 400)),
    transforms.Resize(size=(640, 640)),
    transforms.GaussianBlur(kernel_size=(3, 3), sigma=(0.1, 2.0)),
    transforms.RandomErasing(p=0.3, scale=(0.02, 0.2), ratio=(0.3, 3.3))
])

# 경로 설정
input_dir = 'dataset/images/bottle-good'
output_dir = 'dataset/images/bottle-good-augmented'
os.makedirs(output_dir, exist_ok=True)

# 증강 이미지 생성
for img_name in tqdm(os.listdir(input_dir)):
    img_path = os.path.join(input_dir, img_name)
    img = Image.open(img_path).convert("RGB")
    for i in range(5):  # 5배 증강
        aug_img = bottle_good_aug(img)
        aug_img.save(os.path.join(output_dir, f"{img_name.split('.')[0]}_aug{i}.jpg"))

# 주의: YOLO는 이미지와 동일한 이름의 .txt 라벨 파일이 필요합니다. 증강된 이미지에 맞춰 라벨도 복사해야 합니다.

# ===================================== #
# 2단계: 라벨 파일 복사
# ===================================== #

label_input_dir = 'dataset/labels/bottle-good'
label_output_dir = 'dataset/labels/bottle-good-augmented'
os.makedirs(label_output_dir, exist_ok=True)

for img_name in os.listdir(input_dir):
    label_name = img_name.replace('.jpg', '.txt')
    label_path = os.path.join(label_input_dir, label_name)
    for i in range(5):
        new_label_name = f"{img_name.split('.')[0]}_aug{i}.txt"
        new_label_path = os.path.join(label_output_dir, new_label_name)
        with open(label_path, 'r') as f:
            label_data = f.read()
        with open(new_label_path, 'w') as f:
            f.write(label_data)
